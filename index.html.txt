    // STATISTICAL CALCULATIONS
    function calculatePearson(data) {
      const n = data.length;
      if (n < 2) return null;

      let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
      data.forEach(d => {
        sumX += d.hours;
        sumY += d.score;
        sumXY += (d.hours * d.score);
        sumX2 += (d.hours * d.hours);
        sumY2 += (d.score * d.score);
      });

      const meanX = sumX / n;
      const meanY = sumY / n;

      const num = (n * sumXY) - (sumX * sumY);
      const den = Math.sqrt(((n * sumX2) - (sumX * sumX)) * ((n * sumY2) - (sumY * sumY)));

      const r = den === 0 ? 0 : num / den;

      // Linear regression: y = mx + b
      const slope = ((n * sumX2) - (sumX * sumX)) === 0 ? 0 : num / ((n * sumX2) - (sumX * sumX));
      const intercept = meanY - (slope * meanX);

      return { n, meanX, meanY, r, slope, intercept };
    }

    function classifyCorrelation(r) {
      const absR = Math.abs(r);
      let strength = "";
      if (absR >= 0.9) strength = "Very Strong";
      else if (absR >= 0.7) strength = "Strong";
      else if (absR >= 0.5) strength = "Moderate";
      else if (absR >= 0.3) strength = "Weak";
      else strength = "No / Very Weak";

      const dir = r > 0 ? "Positive" : r < 0 ? "Negative" : "";
      return `${strength} ${dir} Relationship`.trim();
    }

    function runAnalysis() {
      const errBox = document.getElementById('error-container');
      errBox.innerHTML = '';

      if (students.length < 2) {
        errBox.innerHTML = `<div class="p-4 bg-red-50 text-red-700 text-sm rounded-xl mb-4">Please enter at least 2 valid student records to analyze.</div>`;
        return;
      }

      const res = calculatePearson(students);
      if (!res) return;

      currentAnalysis = {
        id: 'analysis-' + Date.now(),
        date: new Date().toLocaleString(),
        n: res.n,
        meanX: res.meanX,
        meanY: res.meanY,
        r: res.r,
        slope: res.slope,
        intercept: res.intercept,
        classification: classifyCorrelation(res.r),
        dataset: JSON.parse(JSON.stringify(students))
      };

      document.getElementById('stat-n').innerText = res.n;
      document.getElementById('stat-mean-x').innerText = res.meanX.toFixed(2) + ' hrs';
      document.getElementById('stat-mean-y').innerText = res.meanY.toFixed(2) + '%';
      document.getElementById('stat-r').innerText = res.r.toFixed(4);

      document.getElementById('res-classification').innerText = currentAnalysis.classification;
      document.getElementById('res-r-val').innerText = (res.r >= 0 ? '+' : '') + res.r.toFixed(2);
      document.getElementById('res-explanation').innerText = `The analysis indicates a ${currentAnalysis.classification.toLowerCase()} between study time and academic scores in this dataset (r = ${res.r.toFixed(4)}). Students with higher study time generally achieved higher academic scores.`;

      drawScatterPlot('scatterCanvas', students, res.slope, res.intercept);
    }

    function drawScatterPlot(canvasId, data, slope, intercept) {
      const canvas = document.getElementById(canvasId);
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      const padding = 50;
      const width = canvas.width - padding * 2;
      const height = canvas.height - padding * 2;

      let maxX = Math.max(...data.map(d => d.hours), 6);
      let maxY = 100;

      // Draw Grid & Axes
      ctx.strokeStyle = '#e2e8f0';
      ctx.lineWidth = 1;
      ctx.beginPath();
      for (let i = 0; i <= 5; i++) {
        const y = padding + (height / 5) * i;
        ctx.moveTo(padding, y);
        ctx.lineTo(padding + width, y);

        ctx.fillStyle = '#94a3b8';
        ctx.font = '10px sans-serif';
        ctx.fillText((100 - (20 * i)).toString(), 15, y + 4);
      }
      ctx.stroke();

      // Axis labels
      ctx.fillStyle = '#475569';
      ctx.font = '12px sans-serif';
      ctx.fillText('Study Time (Hours/Day)', canvas.width / 2 - 50, canvas.height - 10);

      // Plot Points
      data.forEach(d => {
        const x = padding + (d.hours / maxX) * width;
        const y = padding + height - (d.score / maxY) * height;

        ctx.beginPath();
        ctx.arc(x, y, 6, 0, Math.PI * 2);
        ctx.fillStyle = '#4f46e5';
        ctx.fill();
        ctx.strokeStyle = '#ffffff';
        ctx.lineWidth = 2;
        ctx.stroke();
      });

      // Regression Line
      const y1 = intercept;
      const y2 = slope * maxX + intercept;

      const x1Canvas = padding;
      const y1Canvas = padding + height - (Math.min(Math.max(y1, 0), 100) / maxY) * height;
      const x2Canvas = padding + width;
      const y2Canvas = padding + height - (Math.min(Math.max(y2, 0), 100) / maxY) * height;

      ctx.beginPath();
      ctx.moveTo(x1Canvas, y1Canvas);
      ctx.lineTo(x2Canvas, y2Canvas);
      ctx.strokeStyle = '#ef4444';
      ctx.lineWidth = 2.5;
      ctx.stroke();
    }

      let saved = JSON.parse(localStorage.getItem('study_reports') || '[]');
      saved.push(currentAnalysis);

    }

    function deleteSingleReport(id) {
      let saved = JSON.parse(localStorage.getItem('study_reports') || '[]');

      localStorage.setItem('study_reports', JSON.stringify(saved));
      
      compareIds = compareIds.filter(cId => cId !== id);
      renderSavedReportsPage();
    }

      compareIds = [];
      renderSavedReportsPage();

    // COMPARISON PAGE MANAGEMENT
    function toggleCompare(id) {
      if (compareIds.includes(id)) {
        compareIds = compareIds.filter(cId => cId !== id);
      } else {
        if (compareIds.length >= 2) {
          compareIds.shift();
        }
        compareIds.push(id);
      }
      renderComparisonPage();
    }

    function renderComparisonPage() {
      const saved = JSON.parse(localStorage.getItem('study_reports') || '[]');
      const grid = document.getElementById('comparison-selection-grid');
      grid.innerHTML = '';

      if (saved.length === 0) {
        grid.innerHTML = `<p class="text-sm text-slate-400 col-span-full">No saved reports available to compare. Please save analyses from the Data Analysis tab first.</p>`;
      }

      saved.forEach((rep, index) => {
        const isCompared = compareIds.includes(rep.id);
        const card = document.createElement('div');
        card.className = `p-4 rounded-xl border transition-all ${isCompared ? 'bg-indigo-50/50 border-indigo-300' : 'bg-slate-50 border-slate-200'}`;
        card.innerHTML = `
          <div class="flex justify-between items-start mb-2">
            <div>
              <h4 class="font-bold text-slate-900 text-sm">Study Analysis #${index + 1}</h4>
              <p class="text-xs text-slate-400">${rep.date}</p>
            </div>
            <span class="text-xs font-bold text-indigo-600 bg-white px-2 py-0.5 rounded border border-slate-200">r = ${rep.r.toFixed(2)}</span>
          </div>
          <p class="text-xs text-slate-600 mb-3">Sample: ${rep.n} students &bull; ${rep.classification}</p>
          <button onclick="toggleCompare('${rep.id}')" class="w-full py-1.5 text-xs font-medium rounded ${isCompared ? 'bg-indigo-600 text-white' : 'bg-white border border-slate-200 text-slate-700 hover:bg-slate-100'}">
            ${isCompared ? '✓ Selected for Comparison' : '+ Select for Comparison'}
          </button>
        `;
        grid.appendChild(card);
      });

      // Render Comparison Matrix
      const compSection = document.getElementById('comparison-section');
      if (compareIds.length === 2) {
        const rep1 = saved.find(r => r.id === compareIds[0]);
        const rep2 = saved.find(r => r.id === compareIds[1]);

        if (rep1 && rep2) {
          compSection.classList.remove('hidden');
          document.getElementById('comp-head-1').innerText = rep1.date.split(',')[0];
          document.getElementById('comp-head-2').innerText = rep2.date.split(',')[0];

          const tbody = document.getElementById('comparison-table-body');
          tbody.innerHTML = `
            <tr>
              <td class="px-4 py-2 font-medium">Students (N)</td>
              <td class="px-4 py-2">${rep1.n}</td>
              <td class="px-4 py-2">${rep2.n}</td>
              <td class="px-4 py-2 text-indigo-600 font-bold">${rep2.n - rep1.n}</td>
            </tr>
            <tr>
              <td class="px-4 py-2 font-medium">Mean Study Hours</td>
              <td class="px-4 py-2">${rep1.meanX.toFixed(2)} hrs</td>
              <td class="px-4 py-2">${rep2.meanX.toFixed(2)} hrs</td>
              <td class="px-4 py-2 text-indigo-600 font-bold">${(rep2.meanX - rep1.meanX).toFixed(2)} hrs</td>
            </tr>
            <tr>
              <td class="px-4 py-2 font-medium">Mean Score (%)</td>
              <td class="px-4 py-2">${rep1.meanY.toFixed(2)}%</td>
              <td class="px-4 py-2">${rep2.meanY.toFixed(2)}%</td>
              <td class="px-4 py-2 text-indigo-600 font-bold">${(rep2.meanY - rep1.meanY).toFixed(2)}%</td>
            </tr>
            <tr>
              <td class="px-4 py-2 font-medium">Pearson Coefficient (r)</td>
              <td class="px-4 py-2">${rep1.r.toFixed(4)}</td>
              <td class="px-4 py-2">${rep2.r.toFixed(4)}</td>
              <td class="px-4 py-2 text-indigo-600 font-bold">${(rep2.r - rep1.r).toFixed(4)}</td>
            </tr>
          `;
        }
      } else {
        compSection.classList.add('hidden');
      }
    }
